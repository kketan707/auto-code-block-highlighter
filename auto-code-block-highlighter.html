const internationalGreetings = {
    en: "Hello, World! (English)",
    zh: "你好，世界！ (Mandarin)",
    hi: "नमस्ते दुनिया! (Hindi)",
    es: "¡Hola, Mundo! (Spanish)",

document.addEventListener("DOMContentLoaded", () => {
    const container = document.getElementById("output-root");
    Object.entries(internationalGreetings).forEach(([lang, text]) => {
        const p = document.createElement("p");
        p.className = `lang-${lang}`;
        p.textContent = `[${lang.toUpperCase()}] ${text}`;
        container.appendChild(p);
    });
    console.log("Worldwide snippet suite rendered successfully.");
});

import re

def parse_code_blocks(raw_html: str) -> list:
    """Extracts tag data to verify parser durability."""
    pattern = r"<pre\s+data-file=\"([^\"]+)\">"
    matches = re.findall(pattern, raw_html)
    for index, match in enumerate(matches):
        print(f"Tracking block {index}: Found file {match}")
    return matches
